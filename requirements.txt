    <script>
        let amount = 0; 
        let vaultAsset = 0;
        let isRunning = false; let intervalId = null;
        
        const btn = document.getElementById('engineBtn');
        const amountDisplay = document.getElementById('subsidyAmount');
        const vaultAssetDisplay = document.getElementById('vaultAsset');

        const shopPointsDisplay = document.getElementById('shopPoints');
        const userBadgeText = document.getElementById('userBadgeText');

        /* 💡 데이터 로드 */
        function loadSavedData() {
            const loggedInUser = localStorage.getItem("loggedInUser");
            
            if (loggedInUser) {
                const userData = JSON.parse(localStorage.getItem(`user_${loggedInUser}`));
                if (userData) {
                    amount = parseInt(userData.totalBalance) || 0;
                    vaultAsset = parseInt(userData.vaultAsset) || 0;
                    if (userBadgeText) userBadgeText.innerText = loggedInUser + " 대장님";
                }
            } else {
                amount = 0;
                vaultAsset = 0;
                if (userBadgeText) userBadgeText.innerText = "비회원 기적이";
            }
            updateTotalBalance(false);
        }

        /* 💡 데이터 저장 */
        function saveCurrentData() {
            const loggedInUser = localStorage.getItem("loggedInUser");
            if (loggedInUser) {
                const userData = JSON.parse(localStorage.getItem(`user_${loggedInUser}`)) || {};
                userData.totalBalance = amount;
                userData.vaultAsset = vaultAsset;
                localStorage.setItem(`user_${loggedInUser}`, JSON.stringify(userData));
            }
        }

        /* 💡 회원가입 및 로그인 처리 */
        function handleRegister() {
            const id = document.getElementById("authIdInput").value.trim();
            const pw = document.getElementById("authPwInput").value.trim();

            if (!id || !pw) {
                alert("아이디와 비밀번호를 입력해주세요.");
                return;
            }

            if (localStorage.getItem(`user_${id}`)) {
                alert("이미 존재하는 아이디입니다.");
                return;
            }

            const newUser = {
                username: id,
                password: pw,
                totalBalance: 10000000,
                vaultAsset: 0
            };

            localStorage.setItem(`user_${id}`, JSON.stringify(newUser));
            alert(`🎉 회원가입 완료!\n[${id}]님께 가입 축하 지원금 10,000,000원이 지급되었습니다.`);
        }

        function handleLogin() {
            const id = document.getElementById("authIdInput").value.trim();
            const pw = document.getElementById("authPwInput").value.trim();

            const storedData = localStorage.getItem(`user_${id}`);
            if (!storedData) {
                alert("존재하지 않는 회원 계정입니다.");
                return;
            }

            const userData = JSON.parse(storedData);
            if (userData.password !== pw) {
                alert("비밀번호가 일치하지 않습니다.");
                return;
            }

            localStorage.setItem("loggedInUser", id);
            alert(`🐳 환영합니다, ${id}님!\n개인 자산 데이터가 동기화되었습니다.`);
            
            document.getElementById("authIdInput").value = "";
            document.getElementById("authPwInput").value = "";
            closeAdminModal();
            loadSavedData();
        }

        function handleLogout() {
            localStorage.removeItem("loggedInUser");

            closeAdminModal();
            loadSavedData();
        }

        let ctx = gameCanvas.getContext("2d");

        let paddleHeight = 12, paddleWidth = 65, paddleX; 
        let bricks = [];
        
        const rowColors = [
            { top: "#FF4E50", bottom: "#F9D423" },
            { top: "#FF007F", bottom: "#7928CA" },
            { top: "#00F2FE", bottom: "#4FACFE" },

        function startBrickGameEngine() {
            cancelAnimationFrame(gameAnimationId); if(!gameCanvas) return;
            ballX = gameCanvas.width / 2; 
            ballY = gameCanvas.height - 35; 
            dx = 2; 
            dy = -2; 
            paddleX = (gameCanvas.width - paddleWidth) / 2;

            for (let c = 0; c < 6; c++) { 
                bricks[c] = []; 
                for (let r = 0; r < 4; r++) { 
                    bricks[c][r] = { x: 0, y: 0, status: 1 }; 
                } 
            }
            drawBrickGame();
        }

        function setPaddleX(clientX) { 
            let rect = gameCanvas.getBoundingClientRect(); 
            let relativeX = clientX - rect.left; 
            if(relativeX > 0 && relativeX < gameCanvas.width) paddleX = relativeX - paddleWidth / 2; 
        }

        gameCanvas.addEventListener("touchmove", (e) => setPaddleX(e.touches[0].clientX));
        gameCanvas.addEventListener("mousemove", (e) => setPaddleX(e.clientX));
        
        function drawBrickGame() {
            ctx.clearRect(0, 0, gameCanvas.width, gameCanvas.height);
            
            ctx.beginPath(); 
            ctx.arc(ballX, ballY, ballRadius, 0, Math.PI * 2); 
            ctx.fillStyle = "#FFD700"; 
            ctx.shadowBlur = 8;
            ctx.shadowColor = "#FFD700";
            ctx.fill(); 
            ctx.closePath();
            ctx.shadowBlur = 0;

            ctx.beginPath(); 
            let paddleGradient = ctx.createLinearGradient(paddleX, 0, paddleX + paddleWidth, 0);
            paddleGradient.addColorStop(0, '#FF007F');
            paddleGradient.addColorStop(0.5, '#7928CA');
            paddleGradient.addColorStop(1, '#00F2FE');
            ctx.fillStyle = paddleGradient; 
            ctx.roundRect ? ctx.roundRect(paddleX, gameCanvas.height - paddleHeight - 5, paddleWidth, paddleHeight, 6) : ctx.rect(paddleX, gameCanvas.height - paddleHeight - 5, paddleWidth, paddleHeight);
            ctx.fill(); 
            
            ctx.fillStyle = "rgba(255, 255, 255, 0.6)";
            ctx.fillRect(paddleX + 4, gameCanvas.height - paddleHeight - 4, paddleWidth - 8, 2);
            ctx.closePath();

            for (let c = 0; c < 6; c++) {
                for (let r = 0; r < 4; r++) {
                    if (bricks[c][r] && bricks[c][r].status === 1) {
                        let brickX = c * 46 + 13; 
                        let brickY = r * 18 + 15; 
                        bricks[c][r].x = brickX; 
                        bricks[c][r].y = brickY;

                        ctx.beginPath(); 
                        let brickGradient = ctx.createLinearGradient(brickX, brickY, brickX, brickY + 13);
                        brickGradient.addColorStop(0, rowColors[r].top);
                        brickGradient.addColorStop(1, rowColors[r].bottom);

                        ctx.fillStyle = brickGradient; 
                        ctx.roundRect ? ctx.roundRect(brickX, brickY, 41, 14, 3) : ctx.rect(brickX, brickY, 41, 14);
                        ctx.fill(); 

                        ctx.strokeStyle = "rgba(255,255,255,0.4)";
                        ctx.stroke();
                        ctx.closePath();

                        if (ballX > brickX && ballX < brickX + 41 && ballY > brickY && ballY < brickY + 14) { 
                            dy = -dy; 
                            bricks[c][r].status = 0; 
                            amount += 100; 
                            updateTotalBalance(); 
                        }
                    }
                }
            }

            if (ballX + dx > gameCanvas.width - ballRadius || ballX + dx < ballRadius) dx = -dx;
            if (ballY + dy < ballRadius) dy = -dy;
            else if (ballY + dy > gameCanvas.height - ballRadius - paddleHeight - 5) {
                if (ballX > paddleX && ballX < paddleX + paddleWidth) { 
                    dy = -dy; 
                } 
                else if (ballY + dy > gameCanvas.height - ballRadius) { 
                    cancelAnimationFrame(gameAnimationId); 
                    alert("🧱 게임 오버! 다시 시작합니다."); 
                    startBrickGameEngine(); 
                    return; 
                }
            }
            ballX += dx; ballY += dy; 
            gameAnimationId = requestAnimationFrame(drawBrickGame);
        }

        /* 🍓 2. 성장 퍼즐 (3매칭) 엔진 */
        const PUZZLE_SIZE = 6;
        const puzzleItems = ["🍌", "🍓", "🥕", "🍀"];
        let puzzleBoardData = [];
        let firstSelectedPuzzleCell = null;

        function createPuzzleBoard() {
            const boardDiv = document.getElementById("puzzleBoard");
            if(!boardDiv) return;
            boardDiv.innerHTML = "";

            for (let r = 0; r < PUZZLE_SIZE; r++) {
                puzzleBoardData[r] = [];
                for (let c = 0; c < PUZZLE_SIZE; c++) {
                    puzzleBoardData[r][c] = puzzleItems[Math.floor(Math.random() * puzzleItems.length)];

                    const cell = document.createElement("div");
                    cell.className = "puzzle-cell";
                    cell.innerText = puzzleBoardData[r][c];
                    cell.dataset.row = r;
                    cell.dataset.col = c;
                    cell.addEventListener("click", handlePuzzleClick);
                    boardDiv.appendChild(cell);
                }
            }
            firstSelectedPuzzleCell = null;
        }

        function handlePuzzleClick() {
            const row = Number(this.dataset.row);
            const col = Number(this.dataset.col);

            if (!firstSelectedPuzzleCell) {
                firstSelectedPuzzleCell = { row, col };
                this.style.border = "3px solid #FFEB00";
                return;
            }

            let temp = puzzleBoardData[firstSelectedPuzzleCell.row][firstSelectedPuzzleCell.col];
            puzzleBoardData[firstSelectedPuzzleCell.row][firstSelectedPuzzleCell.col] = puzzleBoardData[row][col];
            puzzleBoardData[row][col] = temp;

            firstSelectedPuzzleCell = null;
            updatePuzzleView();
            checkPuzzleMatches();
        }

        function updatePuzzleView() {
            const cells = document.querySelectorAll(".puzzle-cell");
            cells.forEach(cell => {
                const r = cell.dataset.row;
                const c = cell.dataset.col;
                cell.innerText = puzzleBoardData[r][c];
                cell.style.border = "none";
            });
        }

        function checkPuzzleMatches() {
            let matchedCoords = [];

            for (let r = 0; r < PUZZLE_SIZE; r++) {
                let matchLength = 1;
                for (let c = 1; c < PUZZLE_SIZE; c++) {
                    if (puzzleBoardData[r][c] === puzzleBoardData[r][c - 1]) {
                        matchLength++;
                    } else {
                        if (matchLength >= 3) {
                            for (let i = 0; i < matchLength; i++) {
                                matchedCoords.push({ r: r, c: c - 1 - i });
                            }
                        }
                        matchLength = 1;
                    }
                }
                if (matchLength >= 3) {
                    for (let i = 0; i < matchLength; i++) {
                        matchedCoords.push({ r: r, c: PUZZLE_SIZE - 1 - i });
                    }
                }
            }

            for (let c = 0; c < PUZZLE_SIZE; c++) {
                let matchLength = 1;
                for (let r = 1; r < PUZZLE_SIZE; r++) {
                    if (puzzleBoardData[r][c] === puzzleBoardData[r - 1][c]) {
                        matchLength++;
                    } else {
                        if (matchLength >= 3) {
                            for (let i = 0; i < matchLength; i++) {
                                matchedCoords.push({ r: r - 1 - i, c: c });
                            }
                        }
                        matchLength = 1;
                    }
                }
                if (matchLength >= 3) {
                    for (let i = 0; i < matchLength; i++) {
                        matchedCoords.push({ r: PUZZLE_SIZE - 1 - i, c: c });
                    }
                }
            }

            let uniqueMatches = Array.from(new Set(matchedCoords.map(item => JSON.stringify(item)))).map(string => JSON.parse(string));

            if (uniqueMatches.length > 0) {
                uniqueMatches.forEach(coord => {
                    let newEmoji;
                    do {
                        newEmoji = puzzleItems[Math.floor(Math.random() * puzzleItems.length)];
                    } while (newEmoji === puzzleBoardData[coord.r][coord.c]);
                    
                    puzzleBoardData[coord.r][coord.c] = newEmoji;
                });

                let earnedPoints = uniqueMatches.length * 100;
                amount += earnedPoints;
                updateTotalBalance();
                updatePuzzleView();
            }
        }

        /* 🎰 카지노 축하 음악 재생 함수 (Web Audio API 활용) */
        let casinoAudioCtx = null;
        function playCasinoCelebrationMusic() {
            try {
                window.AudioContext = window.AudioContext || window.webkitAudioContext;
                casinoAudioCtx = new AudioContext();

                const notes = [523.25, 659.25, 783.99, 1046.50, 783.99, 1046.50];
                const times = [0, 0.15, 0.3, 0.45, 0.65, 0.8];
                const durations = [0.12, 0.12, 0.12, 0.18, 0.12, 0.4];

                notes.forEach((freq, idx) => {
                    let osc = casinoAudioCtx.createOscillator();
                    let gain = casinoAudioCtx.createGain();
                    
                    osc.type = 'triangle';
                    osc.frequency.setValueAtTime(freq, casinoAudioCtx.currentTime + times[idx]);

                    gain.gain.setValueAtTime(0.3, casinoAudioCtx.currentTime + times[idx]);
                    gain.gain.exponentialRampToValueAtTime(0.001, casinoAudioCtx.currentTime + times[idx] + durations[idx]);

                    osc.connect(gain);
                    gain.connect(casinoAudioCtx.destination);

                    osc.start(casinoAudioCtx.currentTime + times[idx]);
                    osc.stop(casinoAudioCtx.currentTime + times[idx] + durations[idx]);
                });
            } catch(e) {
                console.log("Audio play error", e);
            }
        }

        function drawBulletPattern(bx, by) {
            let count = 16;
            let speed = 2.2;
            for(let i=0; i<count; i++) {
                let angle = (Math.PI * 2 / count) * i + (galagaFrame * 0.03);
                gEnemyBullets.push({

                    y: by,
                    dx: Math.cos(angle) * speed,

        function loopGalagaGame() {
            gCtx.fillStyle = "#111118";
            gCtx.fillRect(0, 0, galagaCanvas.width, galagaCanvas.height);
            galagaFrame++;

            gCtx.fillStyle = "#FFFFFF";
            gStars.forEach(star => {
                star.y += star.speed;
                if(star.y > 380) star.y = 0;
                gCtx.fillRect(star.x, star.y, 1.5, 1.5);
            });

            gCtx.fillStyle = "#FF0055"; gCtx.font = "bold 12px monospace"; gCtx.fillText("PI SCORE", 15, 20);
            gCtx.fillStyle = "#FFFFFF"; gCtx.fillText(galagaScore.toString(), 85, 20);

            if(bossEnemy && bossEnemy.alive) {
                gCtx.fillStyle = "#444";
                gCtx.fillRect(80, 28, 160, 6);
                gCtx.fillStyle = "#FF0055";
                gCtx.fillRect(80, 28, (bossEnemy.hp / bossEnemy.maxHp) * 160, 6);

            gCtx.moveTo(gPlayerX, gPlayerY - 12);
            gCtx.lineTo(gPlayerX - 8, gPlayerY + 8);
            gCtx.lineTo(gPlayerX + 8, gPlayerY + 8);
            gCtx.fill();
            gCtx.closePath();

            if(galagaFrame % 6 === 0) {
                gBullets.push({ x: gPlayerX, y: gPlayerY - 12 });
            }

            gCtx.fillStyle = "#00F2FE";
            for(let i = gBullets.length - 1; i >= 0; i--) {
                let b = gBullets[i];
                b.y -= 8;
                gCtx.fillRect(b.x - 1.5, b.y, 3, 8);
                if(b.y < 25) gBullets.splice(i, 1);
            }

            if(bossEnemy && bossEnemy.alive) {
                bossEnemy.angle += 0.03;
                bossEnemy.x = 160 + Math.sin(bossEnemy.angle) * 90;
                
                gCtx.fillStyle = "#7928CA";
                gCtx.beginPath();
                gCtx.arc(bossEnemy.x, bossEnemy.y, 22, 0, Math.PI * 2);
                gCtx.fill();
                gCtx.closePath();
                gCtx.fillStyle = "#FF007F";
                gCtx.beginPath();
                gCtx.arc(bossEnemy.x, bossEnemy.y, 10, 0, Math.PI * 2);
                gCtx.fill();
                gCtx.closePath();

                if(galagaFrame % 45 === 0) {
                    drawBulletPattern(bossEnemy.x, bossEnemy.y);
                }

                for(let i = gBullets.length - 1; i >= 0; i--) {
                    let b = gBullets[i];
                    let dist = Math.hypot(b.x - bossEnemy.x, b.y - bossEnemy.y);
                    if(dist < 22) {
                        gBullets.splice(i, 1);
                        bossEnemy.hp -= 1;
                        galagaScore += 50;
                        amount += 200;
                        updateTotalBalance();

                        if(bossEnemy.hp <= 0) {
                            bossEnemy.alive = false;
                            alert("🎉 보스 격파 성공! 대단합니다!");
                            startGalagaGameEngine();
                            return;
                        }
                    }
                }
            }

            for(let i = gEnemyBullets.length - 1; i >= 0; i--) {
                let eb = gEnemyBullets[i];
                eb.x += eb.dx;
                eb.y += eb.dy;

                gCtx.beginPath();
                gCtx.arc(eb.x, eb.y, 6, 0, Math.PI * 2);
                gCtx.fillStyle = "#FF00FF";
                gCtx.fill();
                gCtx.lineWidth = 1.5;
                gCtx.strokeStyle = "#FFFF00";
                gCtx.stroke();
                gCtx.closePath();

                let pDist = Math.hypot(eb.x - gPlayerX, eb.y - gPlayerY);
                if(pDist < 8) {
                    cancelAnimationFrame(galagaAnimationId);
                    alert("💥 탄막에 피격되었습니다! 다시 도전하세요.");
                    startGalagaGameEngine();
                    return;
                }

                if(eb.x < 0 || eb.x > 320 || eb.y < 0 || eb.y > 380) {
                    gEnemyBullets.splice(i, 1);
                }
            }

            galagaAnimationId = requestAnimationFrame(loopGalagaGame);
        }

        /* ⚡ 5. 새로 추가된 '쏟아지는 보석 회피' 미니게임 엔진 */
        let avoidCanvas = document.getElementById("avoidCanvas");
        let aCtx = avoidCanvas.getContext("2d");
        let avoidAnimationId = null;
        let avoidPlayerX = 160;
        let avoidPlayerY = 340;
        let avoidObstacles = [];
        let avoidGems = [];
        let avoidScore = 0;
        let avoidFrame = 0;

        function startAvoidGameEngine() {
            cancelAnimationFrame(avoidAnimationId);
            if(!avoidCanvas) return;
            avoidPlayerX = 160;
            avoidPlayerY = 340;
            avoidObstacles = [];
            avoidGems = [];
            avoidScore = 0;
            avoidFrame = 0;
            loopAvoidGame();
        }

        function setAvoidPlayerPos(clientX, clientY) {
            let rect = avoidCanvas.getBoundingClientRect();
            let relativeX = clientX - rect.left;
            let relativeY = clientY - rect.top;
            if(relativeX > 15 && relativeX < avoidCanvas.width - 15) avoidPlayerX = relativeX;
            if(relativeY > 30 && relativeY < avoidCanvas.height - 15) avoidPlayerY = relativeY;
        }

        avoidCanvas.addEventListener("touchmove", (e) => { e.preventDefault(); setAvoidPlayerPos(e.touches[0].clientX, e.touches[0].clientY); });
        avoidCanvas.addEventListener("mousemove", (e) => { setAvoidPlayerPos(e.clientX, e.clientY); });

        function loopAvoidGame() {
            aCtx.fillStyle = "#0f172a";
            aCtx.fillRect(0, 0, avoidCanvas.width, avoidCanvas.height);
            avoidFrame++;

            aCtx.fillStyle = "#E67E22"; aCtx.font = "bold 12px monospace"; aCtx.fillText("AVOID SCORE", 15, 20);
            aCtx.fillStyle = "#FFFFFF"; aCtx.fillText(avoidScore.toString(), 110, 20);

            aCtx.fillStyle = "#F1C40F";
            aCtx.beginPath();
            aCtx.arc(avoidPlayerX, avoidPlayerY, 12, 0, Math.PI * 2);
            aCtx.fill();
            aCtx.strokeStyle = "#E67E22";
            aCtx.lineWidth = 2;
            aCtx.stroke();
            aCtx.closePath();

            if(avoidFrame % 25 === 0) {
                avoidObstacles.push({
                    x: Math.random() * (avoidCanvas.width - 30) + 15,
                    y: -20,
                    size: Math.random() * 12 + 10,
                    speed: Math.random() * 2 + 2.5
                });
            }

            if(avoidFrame % 40 === 0) {
                avoidGems.push({
                    x: Math.random() * (avoidCanvas.width - 20) + 10,
                    y: -20,
                    speed: 2
                });
            }

            aCtx.fillStyle = "#E74C3C";
            for(let i = avoidObstacles.length - 1; i >= 0; i--) {
                let obs = avoidObstacles[i];
                obs.y += obs.speed;
                
                aCtx.beginPath();
                aCtx.arc(obs.x, obs.y, obs.size, 0, Math.PI * 2);
                aCtx.fill();
                aCtx.closePath();

                let dist = Math.hypot(obs.x - avoidPlayerX, obs.y - avoidPlayerY);
                if(dist < obs.size + 10) {
                    cancelAnimationFrame(avoidAnimationId);
                    alert("💥 장애물에 부딪혔습니다! 게임 오버.\n획득 점수: " + avoidScore);
                    startAvoidGameEngine();
                    return;
                }

                if(obs.y > avoidCanvas.height + 30) {
                    avoidObstacles.splice(i, 1);
                }
            }

            aCtx.fillStyle = "#2ECC71";
            for(let i = avoidGems.length - 1; i >= 0; i--) {
                let gem = avoidGems[i];
                gem.y += gem.speed;

                aCtx.beginPath();
                aCtx.arc(gem.x, gem.y, 8, 0, Math.PI * 2);
                aCtx.fill();
                aCtx.closePath();

                let dist = Math.hypot(gem.x - avoidPlayerX, gem.y - avoidPlayerY);
                if(dist < 20) {
                    avoidGems.splice(i, 1);
                    avoidScore += 100;
                    amount += 300;
                    updateTotalBalance();
                    continue;
                }

                if(gem.y > avoidCanvas.height + 30) {
                    avoidGems.splice(i, 1);
                }
            }

            avoidAnimationId = requestAnimationFrame(loopAvoidGame);
        }

        window.onload = loadSavedData;
    </script>
